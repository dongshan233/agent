### 1.1 接口层设计

关键点：Agent 的执行过程要用事件「流」给 App。做法是给第 7~8 周的图跑 `app.stream(..., stream_mode="updates")`，把每步增量翻译成第 10 周定义的事件协议。

`server/api.py`：

```python
"""后端入口。运行: uv run uvicorn server.api:app --port 8000"""
import asyncio
import json
import uuid

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from sse_starlette.sse import EventSourceResponse

from agent import build_app          # 第7~8周：构建带 checkpointer 的图
from llm import call_model_stream    # 薄封装：调 LLM 并产出增量文本

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

graph = build_app()                  # 启动时编译一次（内含 SqliteSaver）


class ChatRequest(BaseModel):
    thread_id: str
    message: str


@app.post("/api/chat")
async def chat(req: ChatRequest):
    config = {"configurable": {"thread_id": req.thread_id}}

    async def event_stream():
        # 方式：直接复用第4周的流式调用思路，把图的过程事件化
        # 简化实现：走图的 stream，把节点更新翻译成协议事件
        try:
            async for update in graph.astream(
                {"messages": [("user", req.message)]},
                config=config,
                stream_mode="updates",
            ):
                for node, payload in update.items():
                    if node == "model":
                        # 模型节点：有 tool_calls 就播报，否则交给 finish 流式文本
                        last = payload["messages"][-1]
                        if getattr(last, "tool_calls", None):
                            for tc in last.tool_calls:
                                yield {"data": json.dumps({
                                    "type": "tool_call",
                                    "name": tc["function"]["name"],
                                    "args": json.loads(tc["function"]["arguments"]),
                                }, ensure_ascii=False)}
                        else:
                            async for piece in stream_text(last):
                                yield {"data": json.dumps({"type": "delta", "text": piece},
                                                          ensure_ascii=False)}
                    elif node == "tools":
                        yield {"data": json.dumps({"type": "tool_result_done"},
                                                  ensure_ascii=False)}
            yield {"data": json.dumps({"type": "done", "usage": last_usage()},
                                      ensure_ascii=False)}
        except Exception as e:
            yield {"data": json.dumps({"type": "error", "code": "agent_error",
                                       "message": str(e)}, ensure_ascii=False)}

    return EventSourceResponse(event_stream())


async def stream_text(msg) -> asyncio.AsyncIterator[str]:
    """把模型最终回答拆成增量文本。最简实现：整段按块吐出。
    进阶实现：直接用第4周的 httpx stream 从模型 API 逐 token 取。"""
    text = msg.content or ""
    for i in range(0, len(text), 8):
        yield text[i:i + 8]
        await asyncio.sleep(0.02)


def last_usage() -> dict:
    return {"total_tokens": 0}      # 第12周从 usage 真实统计，先占位
```

**进阶要求（强烈建议完成）**：把 `stream_text` 换成真流式——在 `call_model` 节点里改用第 4 周的 `httpx.stream` 逐 token 解析，用 `asyncio.Queue` 把 token 从节点推到 SSE。这是「感知延迟」的关键优化：整段生成完再吐 vs 边生成边吐，用户体验差距巨大。

先跑通简化版，MVP 验收后再优化——**先端到端，再打磨**。

## 2. 安卓端：最小对话界面

复用你的主场技能：RecyclerView + 协程。MVP 只需要 3 个文件级别的工作。

### 2.1 数据层

```kotlin
// ChatApi.kt —— Retrofit 定义
interface ChatApi {
    @POST("api/chat")
    @Streaming                      // 关键：流式读取响应体，不能用默认缓冲
    suspend fun chat(@Body body: ChatRequest): Response<ResponseBody>
}

data class ChatRequest(val thread_id: String, val message: String)
```

### 2.2 SSE 解析（协程 Flow）

```kotlin
// AgentRepository.kt —— 把 SSE body 解析成事件流
fun chatEvents(request: ChatRequest): Flow<ChatEvent> = flow {
    val resp = api.chat(request)
    val source = resp.body?.source() ?: throw IOException("empty body")
    while (!source.exhausted()) {
        val line = source.readUtf8Line() ?: break
        if (!line.startsWith("data: ")) continue
        val json = JSONObject(line.removePrefix("data: "))
        when (json.optString("type")) {
            "delta"     -> emit(ChatEvent.Delta(json.getString("text")))
            "tool_call" -> emit(ChatEvent.ToolCall(json.getString("name")))
            "interrupt" -> emit(ChatEvent.Interrupt(json.getString("question")))
            "done"      -> emit(ChatEvent.Done(json.optJSONObject("usage")))
            "error"     -> emit(ChatEvent.Error(json.optString("message")))
        }
    }
}.flowOn(Dispatchers.IO)
```

对照第 4 周：Python 里你逐行读 `data: `，这里是同一协议的 Kotlin 逆写。

### 2.3 界面层（要点）

```kotlin
// ViewModel：收集 Flow，增量更新消息列表
fun send(text: String) {
    viewModelScope.launch {
        messages += Message(role = "user", text = text)
        val replyId = messages.size
        messages += Message(role = "assistant", text = "")   // 先占位
        repo.chatEvents(ChatRequest(threadId, text)).collect { ev ->
            when (ev) {
                is ChatEvent.Delta -> {          // 追加到最后一条助手消息并 notifyItemChanged
                    messages[replyId] = messages[replyId].copy(text = messages[replyId].text + ev.text)
                }
                is ChatEvent.ToolCall -> messages[replyId] =
                    messages[replyId].copy(status = "正在调用 ${ev.name}…")
                is ChatEvent.Done -> {}
                is ChatEvent.Error -> messages[replyId] =
                    messages[replyId].copy(status = "出错: ${ev.message}")
                is ChatEvent.Interrupt -> showConfirmDialog(ev)   // 确认后调 resume 接口
            }
        }
    }
}
```

- `thread_id`：`UUID.randomUUID().toString()` 首次生成，存 `SharedPreferences`（≈ 第 8 周概念在端上的落点）
- 基类 URL 注意模拟器访问宿主机用 `http://10.0.2.2:8000`；真机用电脑局域网 IP
- 明文 HTTP 需在 manifest 配 `networkSecurityConfig`（仅开发环境）

## 3. MVP 打通路线（推荐顺序）

1. 后端 curl 自测通过（半小时级）
2. App 发消息 → 收到 `done` 事件（先不等流式渲染，Console 打印即可）
3. `delta` 逐条追加渲染成功
4. `tool_call` 状态提示显示成功
5. 真机演示全链路：天气场景一次跑通
